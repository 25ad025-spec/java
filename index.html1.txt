/* ================= SEARCH ================= */

function searchDonors(){

    let donors=JSON.parse(localStorage.getItem("bloodcareDonors") || "[]");

    let blood=document.getElementById("searchBlood").value;
    let city=document.getElementById("searchCity").value.toLowerCase();

    let filtered=donors.filter(d=>{

        let bloodMatch=!blood || d.blood===blood;
        let cityMatch=!city || d.city.toLowerCase().includes(city);

        return bloodMatch && cityMatch;

    });

    let results=document.getElementById("results");

    if(filtered.length===0){

        results.innerHTML=`
        <div style="
        grid-column:1/-1;
        text-align:center;
        padding:30px;
        color:#7d8798">
        No matching donors found.
        </div>`;

        return;
    }

    results.innerHTML=filtered.map(d=>`

    <div class="donorCard">

        <div class="avatar">${d.blood}</div>

        <div class="donorInfo">

            <h4>${d.name}</h4>

            <p>📍 ${d.city}</p>

            <p>📞 ${d.phone}</p>

        </div>

    </div>

    `).join("");

}
